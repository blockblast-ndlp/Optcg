    <script>
        const PACKS_URL = "https://raw.githubusercontent.com/buhbbl/punk-records/main/english/packs.json";
        const CARDS_BASE_URL = "https://raw.githubusercontent.com/buhbbl/punk-records/main/english/cards/";
        
        let allPacks = {};
        let activePackCards = [];
        let deck = {}; // {card_id: count}
        let leader = null;
        let selectedColors = new Set(); // Multi-select colors

        // Initialize App
        async function init() {
            loadDeckFromStorage();
            await fetchPacks();
            setupEventListeners();
        }

        // Fetch list of packs and populate dropdown
        async function fetchPacks() {
            try {
                const res = await fetch(PACKS_URL);
                if (!res.ok) throw new Error("Failed to load packs");
                allPacks = await res.json();
                
                const packSelector = document.getElementById("pack-selector");
                packSelector.innerHTML = "";
                
                // Group packs for clean presentation (Boosters, Starters, Extra, etc.)
                const packsArray = Object.values(allPacks);
                
                // Set order priority (boosters first, newer first)
                packsArray.sort((a, b) => b.id.localeCompare(a.id));

                packsArray.forEach(p => {
                    const option = document.createElement("option");
                    option.value = p.id;
                    option.innerText = p.raw_title || `${p.code} - ${p.name}`;
                    packSelector.appendChild(option);
                });

                // Auto select OP-01 initially or first item
                const op01Option = packsArray.find(p => p.raw_title.includes("[OP-01]"));
                if (op01Option) {
                    packSelector.value = op01Option.id;
                } else {
                    packSelector.selectedIndex = 0;
                }

                await fetchCardsForSelectedPack();

            } catch (err) {
                console.error(err);
                document.getElementById("card-grid").innerHTML = `
                    <div class="col-span-full py-10 text-center text-red-500 font-bold bg-red-950/20 border border-red-900 rounded-xl">
                        ❌ โหลดชุดการ์ดล้มเหลว กรุณารีเฟรชหน้าเว็บใหม่อีกครั้ง
                    </div>
                `;
            }
        }

        // Fetch cards of selected pack
        async function fetchCardsForSelectedPack() {
            const packId = document.getElementById("pack-selector").value;
            const countLabel = document.getElementById("filtered-count");
            countLabel.innerText = "กำลังดึงข้อมูล...";
            
            try {
                const res = await fetch(`${CARDS_BASE_URL}${packId}.json`);
                if (!res.ok) throw new Error("Failed to fetch set cards");
                activePackCards = await res.json();
                
                filterAndRender();
            } catch (err) {
                console.error(err);
                document.getElementById("card-grid").innerHTML = `
                    <div class="col-span-full py-10 text-center text-red-500 font-bold bg-red-950/20 border border-red-900 rounded-xl">
                        ❌ ไม่สามารถดึงรูปภาพและการ์ดในชุดนี้ได้ชั่วคราว
                    </div>
                `;
            }
        }

        // Apply filters (text search, colors, category)
        function filterAndRender() {
            const query = document.getElementById("search-input").value.toLowerCase().trim();
            const cat = document.getElementById("category-filter").value;
            
            const filtered = activePackCards.filter(card => {
                // Category match
                if (cat !== "ALL" && card.category !== cat) return false;

                // Color match (AND filter if colors are selected)
                if (selectedColors.size > 0) {
                    const hasColor = card.colors.some(col => selectedColors.has(col));
                    if (!hasColor) return false;
                }

                // Text search match (checks name, card ID, effect text, and types)
                if (query !== "") {
                    const id = card.id ? card.id.toLowerCase() : "";
                    const name = card.name ? card.name.toLowerCase() : "";
                    const effect = card.effect ? card.effect.toLowerCase() : "";
                    const types = card.types ? card.types.join(" ").toLowerCase() : "";
                    
                    if (!id.includes(query) && !name.includes(query) && !effect.includes(query) && !types.includes(query)) {
                        return false;
                    }
                }

                return true;
            });

            document.getElementById("filtered-count").innerText = `พบ ${filtered.length} ใบ`;
            renderGrid(filtered);
        }

        // Render card grid with referrerpolicy bypass proxy
        function renderGrid(cards) {
            const grid = document.getElementById("card-grid");
            grid.innerHTML = "";

            if (cards.length === 0) {
                grid.innerHTML = `
                    <div class="col-span-full py-16 text-center text-gray-500 italic">
                        ไม่พบการ์ดตรงตามเงื่อนไขค้นหา 🔍
                    </div>
                `;
                return;
            }

            cards.forEach(card => {
                const isLeader = card.category === "Leader";
                const countInDeck = deck[card.id] || 0;

                // Create proxy image url using weserv to bypass all hotlink protection perfectly
                const cleanImgUrl = card.image ? card.image.split("?")[0] : "";
                const proxiedImgUrl = `https://images.weserv.nl/?url=${encodeURIComponent(cleanImgUrl)}&w=260&error=ssl`;

                const cardDiv = document.createElement("div");
                cardDiv.className = "bg-[#1f1f1f] border border-gray-800 rounded-2xl p-3 flex flex-col justify-between transition duration-200 card-glow shadow-md";
                
                cardDiv.innerHTML = `
                    <div class="space-y-3">
                        <!-- Image Container with loader -->
                        <div class="relative group overflow-hidden rounded-xl bg-neutral-900 aspect-[3/4] flex items-center justify-center">
                            <img src="${proxiedImgUrl}" alt="${card.name}" class="w-full h-full object-contain rounded-xl transition duration-300" onerror="this.onerror=null; this.src='https://asia-en.onepiece-cardgame.com/images/cardlist/card/OP01-001.png';">
                        </div>

                        <!-- Card Info -->
                        <div class="space-y-1">
                            <h4 class="text-xs font-bold truncate text-white" title="${card.name}">${card.name}</h4>
                            <div class="flex justify-between items-center text-[10px] text-gray-400 font-mono">
                                <span>${card.id}</span>
                                <span>${card.category}</span>
                            </div>
                            <!-- Color Badge -->
                            <div class="flex gap-1">
                                ${card.colors.map(col => `
                                    <span class="text-[9px] px-1.5 py-0.5 rounded font-bold ${
                                        col === 'Red' ? 'bg-red-950/80 text-red-400 border border-red-900/40' :
                                        col === 'Blue' ? 'bg-blue-950/80 text-blue-400 border border-blue-900/40' :
                                        col === 'Green' ? 'bg-green-950/80 text-green-400 border border-green-900/40' :
                                        col === 'Yellow' ? 'bg-yellow-950/80 text-yellow-400 border border-yellow-900/40' :
                                        col === 'Black' ? 'bg-gray-900 text-gray-400 border border-gray-800' :
                                        'bg-purple-950/80 text-purple-400 border border-purple-900/40'
                                    } border">${col}</span>
                                `).join("")}
                            </div>
                        </div>
                    </div>

                    <!-- Actions -->
                    <div class="mt-4 pt-2 border-t border-gray-800/60">
                        ${isLeader ? `
                            <button onclick="setAsLeader('${card.id}')" class="w-full bg-yellow-600 hover:bg-yellow-500 text-white font-bold text-xs py-2 rounded-lg transition duration-150">
                                เลือกเป็น Leader 👑
                            </button>
                        ` : `
                            <button onclick="addCardToDeck('${card.id}')" class="w-full bg-red-600 hover:bg-red-500 text-white font-bold text-xs py-2 px-3 rounded-lg transition duration-150 flex justify-between items-center">
                                <span>เพิ่มการ์ด</span>
                                <span class="bg-red-900 text-red-200 px-2 py-0.5 rounded-full text-[9px] font-mono font-bold">${countInDeck}</span>
                            </button>
                        `}
                    </div>
                `;
                
                grid.appendChild(cardDiv);
            });
        }

        // Leader Handler
        window.setAsLeader = function(cardId) {
            // Find card metadata
            const card = activePackCards.find(c => c.id === cardId);
            if (!card) return;
            
            leader = card;
            saveDeckToStorage();
            updateDeckUI();
            filterAndRender();
        };

        window.removeLeader = function() {
            leader = null;
            saveDeckToStorage();
            updateDeckUI();
            filterAndRender();
        };

        // Deck Add/Remove Handlers
        window.addCardToDeck = function(cardId) {
            const currentCount = deck[cardId] || 0;
            const totalCards = Object.values(deck).reduce((a, b) => a + b, 0);

            if (totalCards >= 50) {
                alert("เด็คหลักมีการ์ดครบกำหนด 50 ใบเรียบร้อยแล้ว!");
                return;
            }

            if (currentCount >= 4) {
                alert("ใส่การ์ดใบเดียวกันซ้ำกันได้สูงสุดไม่เกิน 4 ใบ!");
                return;
            }

            deck[cardId] = currentCount + 1;
            saveDeckToStorage();
            updateDeckUI();
            
            // Re-render single element or full grid to update badges
            filterAndRender();
        };

        window.subtractCardFromDeck = function(cardId) {
            if (deck[cardId]) {
                deck[cardId] -= 1;
                if (deck[cardId] <= 0) {
                    delete deck[cardId];
                }
                saveDeckToStorage();
                updateDeckUI();
                filterAndRender();
            }
        };

        // Render Deck List & Verification & Cost Curve
        function updateDeckUI() {
            const listContainer = document.getElementById("deck-items-list");
            listContainer.innerHTML = "";

            const totalQty = Object.values(deck).reduce((a, b) => a + b, 0);
            document.getElementById("deck-qty").innerText = totalQty;
            document.getElementById("deck-progress").style.width = `${(totalQty / 50) * 100}%`;

            // Active Leader Slot
            const leaderDisplay = document.getElementById("leader-display");
            if (leader) {
                const proxiedImgUrl = `https://images.weserv.nl/?url=${encodeURIComponent(leader.image.split("?")[0])}&w=160&error=ssl`;
                leaderDisplay.innerHTML = `
                    <div class="flex gap-4 items-center w-full text-left">
                        <img src="${proxiedImgUrl}" class="w-16 rounded-lg object-contain bg-neutral-900 border border-gray-700 shadow" onerror="this.src='https://asia-en.onepiece-cardgame.com/images/cardlist/card/OP01-001.png'">
                        <div class="flex-1 min-w-0">
                            <h4 class="text-sm font-bold text-yellow-500 truncate">${leader.name}</h4>
                            <p class="text-[10px] text-gray-400 font-mono">${leader.id}</p>
                            <div class="flex gap-1 mt-1">
                                ${leader.colors.map(col => `<span class="text-[9px] px-1 bg-red-950/60 border border-red-900 text-red-400 font-bold rounded">${col}</span>`).join("")}
                            </div>
                        </div>
                        <button onclick="removeLeader()" class="text-xs text-red-500 hover:text-red-400 font-bold border border-red-900/50 bg-red-950/20 px-2 py-1 rounded">เปลี่ยน ❌</button>
                    </div>
                `;
            } else {
                leaderDisplay.innerHTML = `
                    <div class="text-sm text-gray-500 italic">ยังไม่ได้เลือกลีดเดอร์</div>
                    <p class="text-xs text-gray-600 mt-1">กดปุ่ม "เลือกเป็น Leader" ที่การ์ดลีดเดอร์ด้านขวา</p>
                `;
            }

            // Render list of deck cards
            const deckItems = Object.entries(deck);
            if (deckItems.length === 0) {
                listContainer.innerHTML = `<div class="text-center py-8 text-gray-500 italic text-sm">ยังไม่มีการ์ดในเด็คหลัก</div>`;
            } else {
                // Find card info from activePackCards (or fallback if from another set)
                deckItems.forEach(([cardId, count]) => {
                    let card = activePackCards.find(c => c.id === cardId);
                    
                    // Mock card model if not loaded yet
                    if (!card) {
                        card = { id: cardId, name: cardId, colors: ["Red"], cost: 0 };
                    }

                    // Validate card color matches Leader color
                    let isMismatch = false;
                    let validationBadge = "";
                    if (leader) {
                        isMismatch = !card.colors.some(col => leader.colors.includes(col));
                        if (isMismatch) {
                            validationBadge = `<span class="ml-1.5 px-1.5 py-0.5 rounded text-[8px] bg-red-950 text-red-400 font-bold border border-red-900/50">สีไม่ตรง ⚠️</span>`;
                        }
                    }

                    const row = document.createElement("div");
                    row.className = "flex justify-between items-center bg-[#222] px-3 py-2 rounded-xl border border-gray-800/80 hover:border-gray-700/80 transition shadow-inner";
                    row.innerHTML = `
                        <div class="flex-1 min-w-0 pr-2 text-left">
                            <div class="text-xs font-semibold text-white truncate">${card.name}</div>
                            <div class="text-[9px] text-gray-400 font-mono">${cardId} ${validationBadge}</div>
                        </div>
                        <div class="flex items-center gap-2">
                            <button onclick="subtractCardFromDeck('${cardId}')" class="bg-[#1a1a1a] hover:bg-gray-800 text-white rounded-lg w-6 h-6 flex items-center justify-center text-xs font-bold border border-gray-800">➖</button>
                            <span class="text-xs font-bold text-red-500 font-mono w-4 text-center">${count}</span>
                            <button onclick="addCardToDeck('${cardId}')" class="bg-[#1a1a1a] hover:bg-gray-800 text-white rounded-lg w-6 h-6 flex items-center justify-center text-xs font-bold border border-gray-800">➕</button>
                        </div>
                    `;
                    listContainer.appendChild(row);
                });
            }

            // Rules validator messaging
            const validationBox = document.getElementById("validation-box");
            if (!leader) {
                validationBox.innerHTML = `<span>ℹ️</span> <span>กรุณาเลือกลีดเดอร์ เพื่อตรวจสอบความสอดคล้องของสีการ์ดในเด็ค</span>`;
                validationBox.className = "text-xs p-3 rounded-lg bg-yellow-950/30 border border-yellow-900/50 text-yellow-400 flex items-start gap-2";
            } else if (totalQty < 50) {
                validationBox.innerHTML = `<span>ℹ️</span> <span>เด็คยังขาดการ์ดอีก <strong>${50 - totalQty} ใบ</strong> เพื่อให้ครบ 50 ใบพอดี</span>`;
                validationBox.className = "text-xs p-3 rounded-lg bg-yellow-950/30 border border-yellow-900/50 text-yellow-400 flex items-start gap-2";
            } else if (totalQty === 50) {
                // Check if any card has color mismatch
                let hasColorError = false;
                Object.keys(deck).forEach(cid => {
                    const card = activePackCards.find(c => c.id === cid);
                    if (card) {
                        const match = card.colors.some(col => leader.colors.includes(col));
                        if (!match) hasColorError = true;
                    }
                });

                if (hasColorError) {
                    validationBox.innerHTML = `<span>⚠️</span> <span>เด็คครบ 50 ใบแล้ว แต่มี<strong>การ์ดสีไม่ตรงกับลีดเดอร์ของคุณ</strong></span>`;
                    validationBox.className = "text-xs p-3 rounded-lg bg-red-950/30 border border-red-900/50 text-red-400 flex items-start gap-2";
                } else {
                    validationBox.innerHTML = `<span>✅</span> <span>ยอดเยี่ยมมาก! เด็คของคุณถูกต้องตามกฎแข่งและพร้อมนำไปเล่นแล้ว</span>`;
                    validationBox.className = "text-xs p-3 rounded-lg bg-green-950/30 border border-green-900/50 text-green-400 flex items-start gap-2";
                }
            }

            // Render Cost Curve Chart
            renderCostCurve();

            // Export Text area rendering
            const textLines = [];
            if (leader) {
                textLines.push(`1x ${leader.id}`);
            }
            Object.entries(deck).forEach(([id, count]) => {
                textLines.push(`${count}x ${id}`);
            });
            document.getElementById("export-textarea").value = textLines.join("\n");
        }

        // Render cost curve dynamic bars
        function renderCostCurve() {
            const section = document.getElementById("cost-curve-section");
            const container = document.getElementById("curve-chart-container");
            container.innerHTML = "";

            if (Object.keys(deck).length === 0) {
                section.classList.add("hidden");
                return;
            }
            section.classList.remove("hidden");

            // Aggregate counts by cost values (0 to 10)
            const costCounts = Array(11).fill(0);
            Object.entries(deck).forEach(([cardId, count]) => {
                const card = activePackCards.find(c => c.id === cardId);
                if (card) {
                    const cost = card.cost !== null ? Math.min(card.cost, 10) : 0;
                    costCounts[cost] += count;
                }
            });

            const maxQty = Math.max(...costCounts, 1);

            costCounts.forEach((qty, costIndex) => {
                const heightPercent = (qty / maxQty) * 100;
                
                const barWrapper = document.createElement("div");
                barWrapper.className = "flex flex-col items-center flex-1 group relative h-full justify-end";
                barWrapper.innerHTML = `
                    <!-- Tooltip -->
                    <span class="absolute -top-6 bg-red-600 text-white font-mono font-bold text-[9px] px-1 py-0.5 rounded opacity-0 group-hover:opacity-100 transition duration-150 pointer-events-none">${qty} ใบ</span>
                    <!-- Bar -->
                    <div class="bg-gradient-to-t from-red-800 to-red-600 w-3/5 rounded-t-sm transition-all duration-300 shadow-lg" style="height: ${heightPercent}%"></div>
                `;
                container.appendChild(barWrapper);
            });
        }

        // Setup UI Event listeners
        function setupEventListeners() {
            // Dropdown change
            document.getElementById("pack-selector").addEventListener("change", fetchCardsForSelectedPack);
            
            // Search inputs
            document.getElementById("search-input").addEventListener("input", filterAndRender);
            document.getElementById("category-filter").addEventListener("change", filterAndRender);

            // Import Box toggle
            const importBtn = document.getElementById("import-toggle-btn");
            const importBox = document.getElementById("import-box");
            importBtn.addEventListener("click", () => {
                if (importBox.classList.contains("hidden")) {
                    importBox.classList.remove("hidden");
                    importBtn.innerText = "ปิดกล่องนำเข้า ❌";
                } else {
                    importBox.classList.add("hidden");
                    importBtn.innerText = "คลิกเพื่อเปิดกล่องนำเข้า 📥";
                }
            });

            // Handle Import Logic
            document.getElementById("confirm-import-btn").addEventListener("click", () => {
                const val = document.getElementById("import-textarea").value;
                if (!val.trim()) return;

                const lines = val.split("\n");
                const newDeck = {};
                let newLeader = leader;

                lines.forEach(line => {
                    const match = line.trim().match(/^(\d+)x\s*([A-Za-z0-9-_]+)/i);
                    if (match) {
                        const count = parseInt(match[1]);
                        const id = match[2].toUpperCase();
                        
                        // Treat as leader if count is 1 and set/ID structure is leader-like
                        if (count === 1 && (id.endsWith("-001") || id.endsWith("_p1") || id.includes("L"))) {
                            // Quick lookup or fallback placeholder
                            newLeader = { id: id, name: id, colors: ["Red"], image: `https://en.onepiece-cardgame.com/images/cardlist/card/${id}.png` };
                        } else {
                            newDeck[id] = count;
                        }
                    }
                });

                deck = newDeck;
                leader = newLeader;
                saveDeckToStorage();
                updateDeckUI();
                filterAndRender();

                importBox.classList.add("hidden");
                importBtn.innerText = "คลิกเพื่อเปิดกล่องนำเข้า 📥";
                alert("นำเข้าข้อมูลการ์ดเรียบร้อยแล้ว!");
            });

            // Multi color selection buttons
            const colorButtons = document.querySelectorAll(".color-btn");
            colorButtons.forEach(btn => {
                btn.addEventListener("click", () => {
                    const col = btn.getAttribute("data-color");
                    if (selectedColors.has(col)) {
                        selectedColors.delete(col);
                        btn.classList.add("bg-red-950/20", "text-red-500", "border-red-800");
                        btn.classList.remove("bg-red-600", "text-white");
                    } else {
                        selectedColors.add(col);
                        btn.classList.remove("bg-red-950/20", "text-red-500", "border-red-800");
                        btn.classList.add("bg-red-600", "text-white");
                    }
                    filterAndRender();
                });
            });

            // Clear all deck
            document.getElementById("clear-deck-btn").addEventListener("click", () => {
                if (confirm("คุณแน่ใจหรือไม่ว่าต้องการล้างข้อมูลการ์ดในเด็คทั้งหมด?")) {
                    deck = {};
                    leader = null;
                    saveDeckToStorage();
                    updateDeckUI();
                    filterAndRender();
                }
            });
        }

        // Local Storage Helpers
        function saveDeckToStorage() {
            localStorage.setItem("deck_builder_deck", JSON.stringify(deck));
            if (leader) {
                localStorage.setItem("deck_builder_leader", JSON.stringify(leader));
            } else {
                localStorage.removeItem("deck_builder_leader");
            }
        }

        function loadDeckFromStorage() {
            const savedDeck = localStorage.getItem("deck_builder_deck");
            const savedLeader = localStorage.getItem("deck_builder_leader");
            
            if (savedDeck) {
                deck = JSON.parse(savedDeck);
            }
            if (savedLeader) {
                leader = JSON.parse(savedLeader);
            }
        }

        // Launch Core
        init();
    </script>
